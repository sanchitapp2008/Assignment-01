# Declare variables

a = 10
b = 5.5
c = "Hello"
d = True

print("Value of a:", a, "Type:", type(a))
print("Value of b:", b, "Type:", type(b))
print("Value of c:", c, "Type:", type(c))
print("Value of d:", d, "Type:", type(d))

# Type casting example

a = 10
b = 3.5

# Casting
c = float(a)
d = int(b)

print("Integer value:", a)
print("After converting to float:", c)

print("Float value:", b)
print("After converting to integer:", d)



# Logical operators example

a = 10
b = 5

print("AND operator:", a > 5 and b > 2)
print("OR operator:", a > 5 or b > 10)
print("NOT operator:", not(a > 5))
